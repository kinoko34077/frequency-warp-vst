cmake_minimum_required(VERSION 3.25)
project(FrequencyWarpShell VERSION 0.2.0 LANGUAGES C CXX)
if(NOT DEFINED FW_VST3_SDK_DIR OR NOT EXISTS "${FW_VST3_SDK_DIR}/CMakeLists.txt")
  message(FATAL_ERROR "Specify FW_VST3_SDK_DIR as official recursive Steinberg VST3 SDK checkout")
endif()
set(SMTG_ENABLE_VST3_PLUGIN_EXAMPLES OFF CACHE BOOL "" FORCE)
set(SMTG_ENABLE_VST3_HOSTING_EXAMPLES ON CACHE BOOL "" FORCE)
set(SMTG_ENABLE_VSTGUI_SUPPORT OFF CACHE BOOL "" FORCE)
set(SMTG_CREATE_PLUGIN_LINK OFF CACHE BOOL "" FORCE)
set(vst3sdk_SOURCE_DIR "${FW_VST3_SDK_DIR}")
add_subdirectory("${vst3sdk_SOURCE_DIR}" "${CMAKE_CURRENT_BINARY_DIR}/sdk")
smtg_enable_vst3_sdk()
smtg_add_vst3plugin(FrequencyWarpShell
    src/shell.cpp src/delay.h src/parameter_registry.h
    src/win32_engine_editor.h)
# Official Rust Wasmi engine is linked once into installed fixed shell.
# User-provided .fwengine remains an isolated WASM package, never a native DLL.
if(NOT DEFINED FW_NATIVE_ARCHIVE OR NOT EXISTS "${FW_NATIVE_ARCHIVE}")
  message(FATAL_ERROR "Build fwengine-runtime staticlib and supply FW_NATIVE_ARCHIVE")
endif()
add_library(FrequencyWarpNativeRuntime STATIC IMPORTED GLOBAL)
set_target_properties(FrequencyWarpNativeRuntime PROPERTIES
  IMPORTED_LOCATION "${FW_NATIVE_ARCHIVE}")
target_link_libraries(FrequencyWarpShell PRIVATE sdk FrequencyWarpNativeRuntime)
if(WIN32)
  target_link_libraries(FrequencyWarpShell PRIVATE ws2_32 bcrypt userenv ntdll advapi32 user32 gdi32 comctl32 comdlg32 shell32)
elseif(UNIX)
  target_link_libraries(FrequencyWarpShell PRIVATE pthread dl m)
endif()
target_compile_features(FrequencyWarpShell PRIVATE cxx_std_17)
# UI-2: explicit, pinned official WebView2 SDK prototype. This does not
# migrate the VST3 audio shell, and without the SDK the original Win32 editor
# remains the only compiled view. No network download occurs in CMake.
if(WIN32 AND FW_WEBVIEW2_SDK_DIR)
  if(NOT EXISTS "${FW_WEBVIEW2_SDK_DIR}/build/native/include/WebView2.h"
     OR NOT EXISTS "${FW_WEBVIEW2_SDK_DIR}/build/native/x64/WebView2LoaderStatic.lib")
    message(FATAL_ERROR "FW_WEBVIEW2_SDK_DIR must point to unpacked official WebView2 NuGet package")
  endif()
  target_compile_definitions(FrequencyWarpShell PRIVATE FW_ENABLE_WEBVIEW2=1)
  target_include_directories(FrequencyWarpShell PRIVATE
    "${FW_WEBVIEW2_SDK_DIR}/build/native/include")
  target_link_libraries(FrequencyWarpShell PRIVATE
    "${FW_WEBVIEW2_SDK_DIR}/build/native/x64/WebView2LoaderStatic.lib" ole32)
  target_sources(FrequencyWarpShell PRIVATE src/win32_webview2_editor.h)
  # Do not package an incomplete raw web-demo tree: its shared DSP WASM is
  # intentionally generated by Rust, never tracked under web-demo/.
  # Existing web-demo/stage-pages.mjs verifies the complete recursive asset
  # graph and stages dsp_core.wasm from exactly the same Rust core used on Web.
  if(NOT DEFINED FW_WEB_UI_ASSET_DIR OR NOT EXISTS "${FW_WEB_UI_ASSET_DIR}/index.html"
     OR NOT EXISTS "${FW_WEB_UI_ASSET_DIR}/dsp_core.wasm"
     OR NOT EXISTS "${FW_WEB_UI_ASSET_DIR}/realtime.worklet.js")
    message(FATAL_ERROR
      "WebView2 UI requires staged complete site assets: node web-demo/stage-pages.mjs <core wasm>; pass FW_WEB_UI_ASSET_DIR")
  endif()
  # Keep actual site HTML/CSS/JS external to the VST3 binary. Updating files
  # under Resources/ui changes the interface without recompiling DSP or shell.
  add_custom_command(TARGET FrequencyWarpShell POST_BUILD
    COMMAND ${CMAKE_COMMAND} -E make_directory
      "$<TARGET_FILE_DIR:FrequencyWarpShell>/../Resources/ui"
    COMMAND ${CMAKE_COMMAND} -E copy_directory
      "${FW_WEB_UI_ASSET_DIR}"
      "$<TARGET_FILE_DIR:FrequencyWarpShell>/../Resources/ui"
    COMMAND ${CMAKE_COMMAND} -E copy_if_different
      "${CMAKE_CURRENT_SOURCE_DIR}/ui/host-bridge.mjs"
      "$<TARGET_FILE_DIR:FrequencyWarpShell>/../Resources/ui/host-bridge.mjs"
    COMMENT "Bundle replaceable Frequency Warp HTML/CSS/JS with optional WebView2 VST3")
endif()
