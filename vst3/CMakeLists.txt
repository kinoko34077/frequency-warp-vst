cmake_minimum_required(VERSION 3.25)
project(FrequencyWarpShell VERSION 0.2.0 LANGUAGES C CXX)
if(NOT DEFINED FW_VST3_SDK_DIR OR NOT EXISTS "${FW_VST3_SDK_DIR}/CMakeLists.txt")
  message(FATAL_ERROR "Specify FW_VST3_SDK_DIR as official recursive Steinberg VST3 SDK checkout")
endif()
set(SMTG_ENABLE_VST3_PLUGIN_EXAMPLES OFF CACHE BOOL "" FORCE)
set(SMTG_ENABLE_VST3_HOSTING_EXAMPLES ON CACHE BOOL "" FORCE)
set(SMTG_ENABLE_VSTGUI_SUPPORT OFF CACHE BOOL "" FORCE)
set(SMTG_CREATE_PLUGIN_LINK OFF CACHE BOOL "" FORCE)
set(vst3sdk_SOURCE_DIR "${FW_VST3_SDK_DIR}")
add_subdirectory("${vst3sdk_SOURCE_DIR}" "${CMAKE_CURRENT_BINARY_DIR}/sdk")
smtg_enable_vst3_sdk()
smtg_add_vst3plugin(FrequencyWarpShell
    src/shell.cpp src/delay.h src/parameter_registry.h)
# Link verified Rust WASM interpreter statically once into installed VST3.
# The .fwengine package NEVER contains a platform-native DLL.
if(NOT DEFINED FW_NATIVE_ARCHIVE OR NOT EXISTS "${FW_NATIVE_ARCHIVE}")
  message(FATAL_ERROR "Build fwengine-runtime staticlib and supply FW_NATIVE_ARCHIVE")
endif()
add_library(FrequencyWarpNativeRuntime STATIC IMPORTED GLOBAL)
set_target_properties(FrequencyWarpNativeRuntime PROPERTIES
  IMPORTED_LOCATION "${FW_NATIVE_ARCHIVE}")
target_link_libraries(FrequencyWarpShell PRIVATE sdk FrequencyWarpNativeRuntime)
if(WIN32)
  target_link_libraries(FrequencyWarpShell PRIVATE ws2_32 bcrypt userenv ntdll advapi32)
elseif(UNIX)
  target_link_libraries(FrequencyWarpShell PRIVATE pthread dl m)
endif()
target_compile_features(FrequencyWarpShell PRIVATE cxx_std_17)
smtg_target_configure_version_file(FrequencyWarpShell)
enable_testing()
add_executable(fw_shell_delay_test tests/delay_test.cpp)
target_compile_features(fw_shell_delay_test PRIVATE cxx_std_17)
add_test(NAME fw_shell_delay_test COMMAND fw_shell_delay_test)

add_executable(fw_shell_automation_test tests/automation_registry_test.cpp)
target_compile_features(fw_shell_automation_test PRIVATE cxx_std_17)
add_test(NAME fw_shell_automation_test COMMAND fw_shell_automation_test)

add_executable(fw_swap_mailbox_test tests/swap/swap_test.cpp)
target_compile_features(fw_swap_mailbox_test PRIVATE cxx_std_17)
add_test(NAME fw_swap_mailbox_test COMMAND fw_swap_mailbox_test)

add_executable(fw_swap_concurrency_test tests/swap/concurrency_test.cpp)
target_compile_features(fw_swap_concurrency_test PRIVATE cxx_std_17)
add_test(NAME fw_swap_concurrency_test COMMAND fw_swap_concurrency_test)

add_executable(fw_async_pcm_test tests/async/async_pcm_test.cpp)
target_compile_features(fw_async_pcm_test PRIVATE cxx_std_17)
add_test(NAME fw_async_pcm_test COMMAND fw_async_pcm_test)
