cmake_minimum_required(VERSION 3.25)
project(FrequencyWarpNativeBridgeSmoke LANGUAGES CXX)
if(NOT DEFINED FW_NATIVE_ARCHIVE OR NOT EXISTS "${FW_NATIVE_ARCHIVE}")
  message(FATAL_ERROR "Set FW_NATIVE_ARCHIVE from cargo release staticlib output")
endif()
add_library(fw_rust STATIC IMPORTED GLOBAL)
set_target_properties(fw_rust PROPERTIES IMPORTED_LOCATION "${FW_NATIVE_ARCHIVE}")
add_executable(fw_native_bridge_smoke bridge_smoke.cpp)
target_compile_features(fw_native_bridge_smoke PRIVATE cxx_std_17)
target_link_libraries(fw_native_bridge_smoke PRIVATE fw_rust)
if(WIN32)
  target_link_libraries(fw_native_bridge_smoke PRIVATE ws2_32 bcrypt userenv ntdll advapi32)
elseif(UNIX)
  target_link_libraries(fw_native_bridge_smoke PRIVATE pthread dl m)
endif()
